// Initialize BackAppX
const backappx = new BackAppX({
  apiKey: 'YOUR_API_KEY',
  databaseURL: 'YOUR_DB_URL'
});

// Sign up new user
await backappx.auth.createUser({
  email: 'user@example.com',
  password: 'secure_password'
});

// Listen to database changes
backappx.database('users')
  .on('value', (snapshot) => {
    console.log(snapshot.val());
  });
